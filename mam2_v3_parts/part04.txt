local function refreshText()
if multi then
local labels={}
for _,opt in ipairs(options) do
local rv=tostring(realOption(opt))
if selected[rv] then labels[#labels+1]=displayOption(opt) end
end
b.Text="  "..(#labels>0 and table.concat(labels,", ") or "nenhum...")
else
local current=UIState.values[id]
local shown="nenhum..."
for _,opt in ipairs(options) do
if tostring(realOption(opt))==tostring(current) then shown=displayOption(opt);break end
end
b.Text="  "..shown
end
end

popup.Visible=false
arr.Text="▼"
if openedDropdown==popup then openedDropdown=nil end
end
for _,opt in ipairs(options) do
local rv=realOption(opt)
local item=Instance.new("TextButton")
item.Size=UDim2.new(1,0,0,24)
item.BackgroundColor3=Color3.fromRGB(23,23,27)
item.BorderSizePixel=0
item.Font=Enum.Font.Code
item.TextSize=10
item.TextColor3=COLORS.Font
item.TextXAlignment=Enum.TextXAlignment.Left
item.Text="  "..displayOption(opt)
item.AutoButtonColor=false
item.ZIndex=51
item.Parent=popup
corner(item,3)
item.Activated:Connect(function()
if multi then
selected[tostring(rv)]=not selected[tostring(rv)]
local result={}
for _,o in ipairs(options) do
local actual=realOption(o)
if selected[tostring(actual)] then result[#result+1]=actual end
end
UIState.values[id]=result
refreshText()
setControlValue(id,result,false)
else
UIState.values[id]=rv
refreshText()
close()
setControlValue(id,rv,false)
end
end)
end
b.Activated:Connect(function()
if openedDropdown and openedDropdown~=popup then openedDropdown.Visible=false end
popup.Visible=not popup.Visible
if popup.Visible then
openedDropdown=popup
arr.Text="▲"

popup.Position=UDim2.fromOffset(b.AbsolutePosition.X,b.AbsolutePosition.Y+b.AbsoluteSize.Y+2)

else close() end
end)

UIState.controls[id]={Set=function(v)UIState.values[id]=v;refreshText();setControlValue(id,v,true)end}
return r
end
local function addSlider(parentObj,id,label,minV,maxV,default,suffix)
if UIState.values[id]==nil then UIState.values[id]=default end
local r=rowBase(parentObj,44)
local l=textLabel(r,label,11,false)
l.Position=UDim2.fromOffset(3,0)
l.Size=UDim2.new(1,-76,0,18)
local box=Instance.new("TextBox")
box.AnchorPoint=Vector2.new(1,0)
box.Position=UDim2.new(1,-3,0,0)
box.Size=UDim2.fromOffset(66,18)
box.BackgroundColor3=Color3.fromRGB(15,15,17)
box.BorderSizePixel=0
box.Font=Enum.Font.Code
box.TextSize=10
box.TextColor3=COLORS.Font
box.ClearTextOnFocus=false
box.Parent=r
corner(box,3)
stroke(box,COLORS.Outline,0.2)
local bar=Instance.new("Frame")
bar.Position=UDim2.fromOffset(3,27)
bar.Size=UDim2.new(1,-6,0,5)
bar.BackgroundColor3=Color3.fromRGB(55,57,64)
bar.BorderSizePixel=0
bar.Parent=r
corner(bar,3)
local fill=Instance.new("Frame")
fill.BackgroundColor3=COLORS.Accent
fill.BorderSizePixel=0
fill.Parent=bar
corner(fill,3)
local dragging=false
local function paint(v)
v=tonumber(v) or default
v=math.clamp(v,minV,maxV)
UIState.values[id]=v
fill.Size=UDim2.fromScale((v-minV)/(maxV-minV),1)
box.Text=tostring(v)..(suffix or "")
end
paint(UIState.values[id])
local function updateX(x,notifyChange)
local a=math.clamp((x-bar.AbsolutePosition.X)/math.max(1,bar.AbsoluteSize.X),0,1)
local v=minV+(maxV-minV)*a
local step=(maxV-minV)<=10 and .05 or 1
v=math.floor(v/step+.5)*step
paint(v)
if notifyChange then setControlValue(id,v,false) end
end
bar.InputBegan:Connect(function(i)
if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then
dragging=true;updateX(i.Position.X,true)
end
end)
track(UIS.InputChanged:Connect(function(i)
if dragging and (i.UserInputType==Enum.UserInputType.MouseMovement or i.UserInputType==Enum.UserInputType.Touch) then
updateX(i.Position.X,true)
end
end))
track(UIS.InputEnded:Connect(function(i)
if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then dragging=false end
end))
box.Name=id.."_input"
UIState.values[id.."_input"]=tostring(UIState.values[id])
box.FocusLost:Connect(function()
local n=tonumber((box.Text or ""):match("[-%d%.]+"))
if n then
paint(n)
UIState.values[id.."_input"]=tostring(UIState.values[id])
setControlValue(id,UIState.values[id],false)
else
paint(UIState.values[id])
end
end)
UIState.controls[id]={Set=function(v)paint(v);UIState.values[id.."_input"]=tostring(UIState.values[id]);setControlValue(id,UIState.values[id],true)end}
UIState.controls[id.."_input"]={Set=function(v)box.Text=tostring(v);UIState.values[id.."_input"]=tostring(v)end}
return r
end
