-- MAM2_BOULDER_FARM_FUNCTIONAL_V4
pcall(function()
    topbar.Visible = false
    topbar.Active = false
end)
do
    local dragging = false
    local dragInput, startInput, startPos
    headerBar.Active = true
    headerBar.InputBegan:Connect(function(i)
        if UIState.dragLocked then return end
        if i.UserInputType == Enum.UserInputType.MouseButton1
            or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            startInput = i.Position
            startPos = main.Position
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End
                    or i.UserInputState == Enum.UserInputState.Cancel then
                    dragging = false
                end
            end)
        end
    end)
    headerBar.InputChanged:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseMovement
            or i.UserInputType == Enum.UserInputType.Touch then
            dragInput = i
        end
    end)
    track(UIS.InputChanged:Connect(function(i)
        if dragging and i == dragInput then
            local d = i.Position - startInput
            main.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + d.X,
                startPos.Y.Scale,
                startPos.Y.Offset + d.Y
            )
        end
    end))
end
local BF = S.BoulderFarmV4 or {
    enabled = false,
    nearEnabled = false,
    phase = "scan",
    target = nil,
    targetKind = nil,
    mineClock = 0,
    lootClock = 0,
    scanClock = 0,
    noTargetClock = 0,
    promptMissClock = 0,
    lastProgressAt = os.clock(),
    lastHP = nil,
    targets = {},
    stanceDirection = "Down",
    stanceDistance = 5,
    hopOnNoBoulders = false,
    stuckHop = false,
    stuckMinutes = 10,
    hopMode = "Public",
    status = "Parado",
    statusLabel = nil,
}
S.BoulderFarmV4 = BF
local BOULDER_KINDS = {
    mossite = "Mossite",
    voltite = "Voltite",
    gildrite = "Gildrite",
    rimeveil = "Rimeveil",
    nocturnite = "Nocturnite",
}
local function bfSet(v)
    local out = {}
    if type(v) == "string" then
        out[v] = true
    elseif type(v) == "table" then
        for k, x in pairs(v) do
            if x == true then
                out[tostring(k)] = true
            elseif x ~= false and x ~= nil then
                out[tostring(x)] = true
            end
        end
    end
    return out
end
local function bfStatusLabel()
    if BF.statusLabel and BF.statusLabel.Parent then
        return BF.statusLabel
    end
    local page = UIState.pages and UIState.pages["Boulders"]
    if not page then return nil end
    for _, obj in ipairs(page:GetDescendants()) do
        if obj:IsA("TextLabel") and tostring(obj.Text or "") == "Parado" then
            BF.statusLabel = obj
            return obj
        end
    end
end
local function setBFStatus(text)
    BF.status = tostring(text or "")
    local l = bfStatusLabel()
    if l then pcall(function() l.Text = BF.status end) end
end
local function kindFromObject(obj)
    local p = obj
    for _ = 1, 8 do
        if not p or p == Workspace then break end
        local name = tostring(p.Name or ""):lower()
        for token, kind in pairs(BOULDER_KINDS) do
            if name:find(token, 1, true) then return kind end
        end
        for _, attrName in ipairs({"BoulderType", "BoulderKind", "Type", "CrystalName"}) do
            local v = getAttr(p, attrName)
            if v ~= nil then
                local low = tostring(v):lower()
                for token, kind in pairs(BOULDER_KINDS) do
                    if low:find(token, 1, true) then return kind end
                end
            end
        end
        p = p.Parent
    end
end
local function boulderObjectFrom(obj)
    if not obj then return nil, nil end
    local p = obj
    local chosen, kind
    for _ = 1, 8 do
        if not p or p == Workspace then break end
        local k = kindFromObject(p)
        if k then
            kind = k
            chosen = p
            if p:IsA("Model") then break end
        end
        p = p.Parent
    end
    if not chosen or not kind then return nil, nil end
    if chosen:IsA("BasePart") and chosen.Parent and chosen.Parent:IsA("Model") then
        local parentKind = kindFromObject(chosen.Parent)
        if parentKind == kind then chosen = chosen.Parent end
    end
    return chosen, kind
end
local function anchorOfBoulder(target)
    if not target or not target.Parent then return nil end
    if target:IsA("BasePart") then return target end
    if target:IsA("Model") then
        if target.PrimaryPart then return target.PrimaryPart end
        for _, d in ipairs(target:GetDescendants()) do
            if d:IsA("ProximityPrompt") then
                local p = d.Parent
                if p and p:IsA("Attachment") then p = p.Parent end
                if p and p:IsA("BasePart") then return p end
            end
        end
        return target:FindFirstChildWhichIsA("BasePart", true)
    end
end
local function selectedKind(kind)
    if next(BF.targets) == nil then return true end
    return BF.targets[kind] == true
end
local function boulderHealth(target, anchor)
    local names = {"MinedHP", "HP", "Health", "HitPoints"}
    for _, obj in ipairs({target, anchor}) do
        if obj then
            for _, n in ipairs(names) do
                local v = tonumber(getAttr(obj, n))
                if v ~= nil then return v end
            end
        end
    end
    return nil
end
local function findMinePrompt(target)
    if not target or not target.Parent then return nil end
    local prompts = {}
    for _, d in ipairs(target:GetDescendants()) do
        if d:IsA("ProximityPrompt") then prompts[#prompts + 1] = d end
    end
    if #prompts == 0 then return nil end
    local fallback
    for _, p in ipairs(prompts) do
        fallback = fallback or p
        local text = (tostring(p.ActionText or "") .. " " .. tostring(p.ObjectText or "")):lower()
        if text:find("mine", 1, true)
            or text:find("minerar", 1, true)
            or text:find("dig", 1, true)
            or text:find("escavar", 1, true)
            or text:find("break", 1, true)
            or text:find("quebrar", 1, true) then
            return p
        end
    end
    return #prompts == 1 and prompts[1] or fallback
end
local function findPickaxe()
    local c = char()
    local backpack = LP and LP:FindFirstChildOfClass("Backpack")
    local best
    local function consider(tool)
        if not tool or not tool:IsA("Tool") then return end
        local name = tool.Name:lower()
        local score = 0
        if name:find("pickaxe", 1, true) then score += 100 end
        if name:find("pick", 1, true) then score += 40 end
        if getAttr(tool, "SwingSpeed") ~= nil then score += 30 end
        if getAttr(tool, "SwingCooldown") ~= nil then score += 20 end
        if score > 0 and (not best or score > best.score) then
            best = {tool = tool, score = score}
        end
    end
    if c then for _, x in ipairs(c:GetChildren()) do consider(x) end end
    if backpack then for _, x in ipairs(backpack:GetChildren()) do consider(x) end end
    if not best then return nil end
    local tool = best.tool
    if c and tool.Parent ~= c then
        local h = hum()
        if h then pcall(function() h:EquipTool(tool) end) end
    end
    return tool
end
local function requestStreamAt(pos)
    if not pos then return end
    if LP and type(LP.RequestStreamAroundAsync) == "function" then
        task.spawn(function()
            pcall(function() LP:RequestStreamAroundAsync(pos, 1.5) end)
        end)
    end
end
local function safeApproach(anchor, prompt)
    local c = char()
    local r = root()
    if not c or not r or not anchor then return false end
    local maxDist = 10
    if prompt then
        local ok, v = pcall(function() return prompt.MaxActivationDistance end)
        if ok and tonumber(v) then maxDist = math.max(6, tonumber(v)) end
    end
    if (r.Position - anchor.Position).Magnitude <= maxDist * 0.72 then return true end
    requestStreamAt(anchor.Position)
    local rawDist = math.abs(tonumber(BF.stanceDistance) or 5)
    local step = math.clamp(rawDist, 2.5, math.max(3, maxDist * 0.6))
    local dirName = tostring(BF.stanceDirection or "Down")
    local dir
    if dirName == "Front" then
        dir = -anchor.CFrame.LookVector
    elseif dirName == "Back" then
        dir = anchor.CFrame.LookVector
    elseif dirName == "Left" then
        dir = -anchor.CFrame.RightVector
    elseif dirName == "Right" then
        dir = anchor.CFrame.RightVector
    elseif dirName == "Up" then
        dir = Vector3.new(0, 1, 0)
    else
        local flat = Vector3.new(r.Position.X - anchor.Position.X, 0, r.Position.Z - anchor.Position.Z)
        dir = flat.Magnitude > 0.1 and flat.Unit or -anchor.CFrame.LookVector
    end
    local targetPos = anchor.Position + dir * step + Vector3.new(0, 1.5, 0)
    local cf = CFrame.new(targetPos, anchor.Position)
    local ok = pcall(function() c:PivotTo(cf) end)
    if not ok then ok = pcall(function() r.CFrame = cf end) end
    if ok then
        pcall(function()
            r.AssemblyLinearVelocity = Vector3.zero
            r.AssemblyAngularVelocity = Vector3.zero
        end)
    end
    return ok
end
local function activateMining(target, prompt)
    if not target or not target.Parent then return false end
    local tool = findPickaxe()
    if not tool then
        setBFStatus("Sem picareta")
        return false
    end
    local fired = false
    if prompt then
        if type(S.FirePrompt) == "function" then
            local ok, result = pcall(S.FirePrompt, prompt, true)
            fired = ok and result == true
        end
        if not fired and type(fireproximityprompt) == "function" then
            fired = pcall(fireproximityprompt, prompt)
        end
    end
    local activated = pcall(function() tool:Activate() end)
    return fired or activated
end
local function boulderCandidates(maxDistance)
    local r = root()
    if not r then return {} end
    local out, seen = {}, {}
    local maxD = tonumber(maxDistance) or math.huge
    for _, x in ipairs(Workspace:GetDescendants()) do
        if x:IsA("Model") or x:IsA("BasePart") then
            local obj, kind = boulderObjectFrom(x)
            if obj and kind and selectedKind(kind) and not seen[obj] then
                local anchor = anchorOfBoulder(obj)
                if anchor then
                    seen[obj] = true
                    local d = (anchor.Position - r.Position).Magnitude
                    if d <= maxD then
                        out[#out + 1] = {object = obj, kind = kind, anchor = anchor, distance = d}
                    end
                end
            end
        end
    end
    table.sort(out, function(a, b) return a.distance < b.distance end)
    return out
end
local function collectBoulderLoot()
    if type(S.CollectCrystalsBurst) == "function" then
        pcall(S.CollectCrystalsBurst, 35, 12, true)
    end
    if type(S.GrabRunesNear) == "function" then
        pcall(S.GrabRunesNear, 35, 4, true)
    end
end
local function beginBoulderLoot()
    BF.phase = "loot"
    BF.lootClock = 0
    BF.target = nil
    BF.targetKind = nil
    BF.lastHP = nil
    BF.promptMissClock = 0
    setBFStatus("Coletando cristais e Runas")
end
local function mineTargetStep(dt, allowMove)
    local target = BF.target
    if not target or not target.Parent then beginBoulderLoot(); return end
    local anchor = anchorOfBoulder(target)
    if not anchor or not anchor.Parent then beginBoulderLoot(); return end
    local hp = boulderHealth(target, anchor)
    if hp ~= nil then
        if hp <= 0 then beginBoulderLoot(); return end
        if BF.lastHP == nil or hp < BF.lastHP then BF.lastProgressAt = os.clock() end
        BF.lastHP = hp
    end
    local prompt = findMinePrompt(target)
    if allowMove then safeApproach(anchor, prompt) end
    BF.mineClock += dt
    local swingGap = 0.12
    local tool = findPickaxe()
    if tool then
        local observed = tonumber(getAttr(tool, "SwingCooldown") or getAttr(tool, "SwingSpeed"))
        if observed and observed > 0 and observed <= 3 then
            swingGap = math.max(0.02, observed * 0.4)
        end
    end
    if BF.mineClock >= swingGap then
        BF.mineClock = 0
        local acted = activateMining(target, prompt)
        if acted then BF.promptMissClock = 0 else BF.promptMissClock += swingGap end
    end
    if type(S.CollectCrystalsBurst) == "function" then
        pcall(S.CollectCrystalsBurst, 25, 6, true)
    end
    if type(S.GrabRunesNear) == "function" then
        pcall(S.GrabRunesNear, 20, 4, true)
    end
    local kind = BF.targetKind or kindFromObject(target) or "Boulder"
    if hp ~= nil then
        setBFStatus(string.format("Minerando %s  %.0f HP", kind, hp))
    else
        setBFStatus("Minerando " .. kind)
    end
    if BF.stuckHop and (os.clock() - BF.lastProgressAt) >= math.max(60, BF.stuckMinutes * 60) then
        setBFStatus("Travado - trocando de servidor")
        BF.enabled = false
        S.BoulderLootAssist = false
        hopServer(tostring(BF.hopMode) == "Lowest Server")
    end
end
local function selectNewTarget()
    local item = boulderCandidates()[1]
    if not item then return false end
    BF.target = item.object
    BF.targetKind = item.kind
    BF.phase = "mine"
    BF.mineClock = 0
    BF.lastHP = boulderHealth(item.object, item.anchor)
    BF.lastProgressAt = os.clock()
    BF.promptMissClock = 0
    requestStreamAt(item.anchor.Position)
    setBFStatus("Indo para " .. item.kind)
    return true
end
local function resetBoulderFarm()
    BF.phase = "scan"
    BF.target = nil
    BF.targetKind = nil
    BF.mineClock = 0
    BF.lootClock = 0
    BF.scanClock = 0
    BF.noTargetClock = 0
    BF.promptMissClock = 0
    BF.lastHP = nil
    BF.lastProgressAt = os.clock()
end
KNOWN_TOGGLES.AutoFarmBoulders = function(v)
    BF.enabled = v == true
    S.BoulderLootAssist = BF.enabled
    resetBoulderFarm()
    setBFStatus(BF.enabled and "Procurando Boulder" or "Parado")
end
KNOWN_TOGGLES.AutoNearCollect = function(v)
    BF.nearEnabled = v == true
end
KNOWN_OPTIONS.FarmTargets = function(v) BF.targets = bfSet(v) end
KNOWN_OPTIONS.BoulderStanceDirection = function(v) BF.stanceDirection = tostring(v or "Down") end
KNOWN_OPTIONS.BoulderStanceDistance = function(v) BF.stanceDistance = tonumber(v) or 5 end
KNOWN_TOGGLES.HopOnNoBoulders = function(v) BF.hopOnNoBoulders = v == true end
KNOWN_TOGGLES.StuckHop = function(v) BF.stuckHop = v == true end
KNOWN_OPTIONS.StuckHopMinutes = function(v) BF.stuckMinutes = math.clamp(tonumber(v) or 10, 1, 60) end
KNOWN_OPTIONS.HopServerMode = function(v) BF.hopMode = tostring(v or "Public") end
BF.targets = bfSet(UIState.values.FarmTargets or {"Nocturnite"})
BF.stanceDirection = tostring(UIState.values.BoulderStanceDirection or "Down")
BF.stanceDistance = tonumber(UIState.values.BoulderStanceDistance) or 5
BF.hopOnNoBoulders = UIState.values.HopOnNoBoulders == true
BF.stuckHop = UIState.values.StuckHop == true
BF.stuckMinutes = tonumber(UIState.values.StuckHopMinutes) or 10
BF.hopMode = tostring(UIState.values.HopServerMode or "Public")
track(UIS.InputBegan:Connect(function(i, processed)
    if processed or UIS:GetFocusedTextBox() or i.UserInputType ~= Enum.UserInputType.Keyboard then return end
    local bKey = tostring(UIState.values.AutoFarmBouldersKey or "None")
    if bKey ~= "None" and i.KeyCode.Name == bKey then
        local nextValue = not BF.enabled
        UIState.values.AutoFarmBoulders = nextValue
        KNOWN_TOGGLES.AutoFarmBoulders(nextValue)
    end
    local nKey = tostring(UIState.values.AutoNearCollectKey or "None")
    if nKey ~= "None" and i.KeyCode.Name == nKey then
        local nextValue = not BF.nearEnabled
        UIState.values.AutoNearCollect = nextValue
        KNOWN_TOGGLES.AutoNearCollect(nextValue)
    end
end))
track(RunService.Heartbeat:Connect(function(dt)
    if S.Destroyed then return end
    if BF.enabled then
        if BF.phase == "scan" then
            BF.scanClock += dt
            if BF.scanClock >= 0.8 then
                BF.scanClock = 0
                if selectNewTarget() then
                    BF.noTargetClock = 0
                else
                    BF.noTargetClock += 0.8
                    setBFStatus("Procurando Boulder")
                    collectBoulderLoot()
                    if BF.hopOnNoBoulders and BF.noTargetClock >= 8 then
                        BF.enabled = false
                        S.BoulderLootAssist = false
                        setBFStatus("Sem Boulders - trocando servidor")
                        hopServer(tostring(BF.hopMode) == "Lowest Server")
                    end
                end
            end
        elseif BF.phase == "mine" then
            mineTargetStep(dt, true)
        elseif BF.phase == "loot" then
            BF.lootClock += dt
            collectBoulderLoot()
            if BF.lootClock >= 1.6 then
                BF.phase = "reset"
                BF.lootClock = 0
                setBFStatus("Runas coletadas")
            end
        elseif BF.phase == "reset" then
            BF.phase = "scan"
            BF.scanClock = 0
            BF.noTargetClock = 0
            setBFStatus("Próximo Boulder")
        else
            resetBoulderFarm()
        end
    end
    if BF.nearEnabled and not BF.enabled then
        BF.scanClock += dt
        if BF.scanClock >= 0.2 then
            BF.scanClock = 0
            local item = boulderCandidates(35)[1]
            if item then
                BF.target = item.object
                BF.targetKind = item.kind
                mineTargetStep(0.2, false)
            end
        end
    end
end))
local previousDestroyBoulderV4 = S.Destroy
S.Destroy = function()
    BF.enabled = false
    BF.nearEnabled = false
    S.BoulderLootAssist = false
    pcall(function() if topbar then topbar.Visible = false end end)
    if previousDestroyBoulderV4 then return previousDestroyBoulderV4() end
end
