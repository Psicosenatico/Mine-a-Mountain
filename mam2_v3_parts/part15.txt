
-- MAM2_AUDIT_FINALIZATION_V3_1
-- Finalização da segunda auditoria funcional.
-- Corrige a unidade de PkDelay (UI em ms) e substitui a varredura global
-- do Auto Rune Pickup por uma busca restrita aos containers de runas.

S.AuditFinalV31 = S.AuditFinalV31 or {
    autoRune = false,
    mode = UIState.values.AutoRunePickupMode or {"Near"},
    clock = 0,
}
local AF = S.AuditFinalV31

-- ---------------------------------------------------------------------------
-- Auto Pickup / PkDelay
-- A interface original expõe PkDelay em milissegundos (0..400; padrão 120ms).
-- O core usa segundos. A revisão anterior estava aplicando o valor da UI
-- diretamente como segundos; esta camada normaliza explicitamente ms -> s.
-- ---------------------------------------------------------------------------
local function setPickupDelayMs(v)
    local ms = tonumber(v)
    if ms == nil then ms = 120 end
    ms = math.clamp(ms, 0, 400)
    S.PICK.cooldown = math.clamp(ms / 1000, 0.02, 0.4)
end

KNOWN_OPTIONS.PkDelay = setPickupDelayMs
setPickupDelayMs(UIState.values.PkDelay)

-- ---------------------------------------------------------------------------
-- Auto Rune Pickup
-- Desliga a implementação V3 anterior antes de instalar o binding final.
-- Isso mantém o Heartbeat antigo inerte e evita dupla coleta/teleporte.
-- ---------------------------------------------------------------------------
local legacyAutoRuneSetter = KNOWN_TOGGLES.AutoRunePickup
if type(legacyAutoRuneSetter) == "function" then
    pcall(legacyAutoRuneSetter, false)
end
S.AutoRunePickup = false -- também mantém inerte o loop compartilhado do part10.

local function runeModeHas(name)
    local v = AF.mode
    if type(v) == "string" then
        return v == name
    end
    if type(v) == "table" then
        for k, x in pairs(v) do
            if x == true and tostring(k) == name then return true end
            if tostring(x) == name then return true end
        end
    end
    return false
end

local function findRuneContainer()
    local direct = Workspace:FindFirstChild("DroppedRunes")
        or Workspace:FindFirstChild("Runes")
        or Workspace:FindFirstChild("Dropped Rune")
    if direct then return direct end

    -- Fallback restrito: procura um container nomeado, sem classificar o mapa
    -- inteiro como runa a cada Heartbeat.
    for _, obj in ipairs(Workspace:GetChildren()) do
        local n = obj.Name:lower()
        if n == "droppedrunes" or n == "runes" or n == "dropped rune" then
            return obj
        end
    end
end

local function runePartFromPrompt(prompt)
    if not prompt or not prompt.Parent then return nil end
    local p = prompt.Parent
    if p:IsA("BasePart") then return p end
    if p:IsA("Attachment") and p.Parent and p.Parent:IsA("BasePart") then
        return p.Parent
    end
    local model = p:FindFirstAncestorOfClass("Model")
    if model then
        return model.PrimaryPart or model:FindFirstChildWhichIsA("BasePart", true)
    end
    return p:FindFirstAncestorWhichIsA("BasePart")
end

local function listRunePrompts(origin, maxDistance)
    local folder = findRuneContainer()
    if not folder then return {} end

    local out = {}
    for _, obj in ipairs(folder:GetDescendants()) do
        if obj:IsA("ProximityPrompt") and obj.Parent then
            local part = runePartFromPrompt(obj)
            if part then
                local distance = (part.Position - origin).Magnitude
                if distance <= maxDistance then
                    out[#out + 1] = {
                        prompt = obj,
                        part = part,
                        distance = distance,
                    }
                end
            end
        end
    end

    table.sort(out, function(a, b)
        return a.distance < b.distance
    end)
    return out
end

local function fireRunePrompt(prompt)
    if not prompt or not prompt.Parent then return false end
    if S.FirePrompt then
        local ok, result = pcall(S.FirePrompt, prompt, true)
        if ok and result == true then return true end
    end
    if type(fireproximityprompt) == "function" then
        return pcall(fireproximityprompt, prompt)
    end
    return false
end

local function pivotRuneTarget(part)
    local c = char()
    local r = root()
    if not c or not r or not part or not part.Parent then return false end

    local target = part.Position + Vector3.new(0, 3, 0)
    local cf = CFrame.new(target, part.Position)
    local ok = pcall(function()
        c:PivotTo(cf)
    end)
    if not ok then
        ok = pcall(function()
            r.CFrame = cf
        end)
    end
    if ok then
        pcall(function()
            r.AssemblyLinearVelocity = Vector3.zero
            r.AssemblyAngularVelocity = Vector3.zero
        end)
    end
    return ok
end

local function grabNearFinal()
    -- Reutiliza o core já reconstruído, que limita a 4 prompts e usa raio 20.
    if type(S.GrabRunesNear) == "function" then
        local ok, grabbed, seen = pcall(S.GrabRunesNear, 20, 4, true)
        if ok then
            return tonumber(grabbed) or 0, tonumber(seen) or 0
        end
    end

    local r = root()
    if not r then return 0, 0 end
    local list = listRunePrompts(r.Position, 20)
    local grabbed = 0
    for i = 1, math.min(4, #list) do
        if fireRunePrompt(list[i].prompt) then grabbed += 1 end
    end
    return grabbed, #list
end

local function grabOneMapWide()
    local r = root()
    if not r then return false end
    local list = listRunePrompts(r.Position, 5000)
    local item = list[1]
    if not item then return false end

    if not pivotRuneTarget(item.part) then return false end
    return fireRunePrompt(item.prompt)
end

KNOWN_TOGGLES.AutoRunePickup = function(v)
    AF.autoRune = v == true
    S.AutoRunePickup = false
end

KNOWN_OPTIONS.AutoRunePickupMode = function(v)
    AF.mode = v
    UIState.values.AutoRunePickupMode = v
end

AF.mode = UIState.values.AutoRunePickupMode or AF.mode

track(RunService.Heartbeat:Connect(function(dt)
    if S.Destroyed or not AF.autoRune then return end

    AF.clock += dt
    if AF.clock < 0.2 then return end
    AF.clock = 0

    local near = runeModeHas("Near")
    local teleport = runeModeHas("Teleport")
    local both = runeModeHas("Both")

    if near or both then
        local _, seen = grabNearFinal()
        -- "Both" só teleporta quando não existe mais nada próximo.
        if near or seen > 0 then return end
    end

    if teleport or both then
        grabOneMapWide()
    end
end))

-- Cleanup complementar. O setter antigo continua desligado por toda a sessão.
local previousDestroyFinal = S.Destroy
S.Destroy = function()
    AF.autoRune = false
    S.AutoRunePickup = false
    if type(legacyAutoRuneSetter) == "function" then
        pcall(legacyAutoRuneSetter, false)
    end
    if previousDestroyFinal then
        return previousDestroyFinal()
    end
end

notify("Auditoria V3.1 concluída: PkDelay corrigido e Auto Rune otimizado.", 4)
