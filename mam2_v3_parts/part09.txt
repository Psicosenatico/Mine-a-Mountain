local function ptbrText(value)
    if type(value)~="string" or value=="" then return value end
    local exact=PTBR_EXACT[value]
    if exact then return exact end
    local out=value
    for _,pair in ipairs(PTBR_PHRASES) do
        out=out:gsub(pair[1],pair[2])
    end
    return out
end
local translating=setmetatable({},{__mode="k"})
local function translateObject(obj)
    if not obj or not obj.Parent then return end
    if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
        if not translating[obj] then
            translating[obj]=true
            pcall(function() obj.Text=ptbrText(obj.Text) end)
            if obj:IsA("TextBox") then
                pcall(function() obj.PlaceholderText=ptbrText(obj.PlaceholderText) end)
            end
            translating[obj]=nil
        end
    end
end
for _,obj in ipairs(gui:GetDescendants()) do translateObject(obj) end
track(gui.DescendantAdded:Connect(function(obj)
    task.defer(function()
        translateObject(obj)
        if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
            track(obj:GetPropertyChangedSignal("Text"):Connect(function()
                if translating[obj] then return end
                local translated=ptbrText(obj.Text)
                if translated~=obj.Text then
                    translating[obj]=true
                    obj.Text=translated
                    translating[obj]=nil
                end
            end))
        end
    end)
end))
pcall(function()
    topTitle.Text="Mine a Mountain 2  |  Versão reconstruída em português"
    headerTime.Text="Tempo: Independente"
end)
