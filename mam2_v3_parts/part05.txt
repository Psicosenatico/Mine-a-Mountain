local function addKey(parentObj,id,label,default)
if UIState.values[id]==nil then UIState.values[id]=default or "None" end
local r=rowBase(parentObj,26)
local l=textLabel(r,label,11,false)
l.Position=UDim2.fromOffset(3,0)
l.Size=UDim2.new(1,-76,1,0)
local b=Instance.new("TextButton")
b.AnchorPoint=Vector2.new(1,.5)
b.Position=UDim2.new(1,-3,.5,0)
b.Size=UDim2.fromOffset(68,21)
b.BackgroundColor3=Color3.fromRGB(15,15,17)
b.BorderSizePixel=0
b.Font=Enum.Font.Code
b.TextSize=10
b.TextColor3=COLORS.Font
b.Text=tostring(UIState.values[id])
b.AutoButtonColor=false
b.Parent=r
corner(b,3);stroke(b,COLORS.Outline,0.2)
b.Activated:Connect(function()
b.Text="..."
local conn
conn=UIS.InputBegan:Connect(function(i,p)
if p then return end
if i.UserInputType==Enum.UserInputType.Keyboard then
UIState.values[id]=i.KeyCode.Name
b.Text=i.KeyCode.Name
conn:Disconnect()
end
end)
end)
return r
end
