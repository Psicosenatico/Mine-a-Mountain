elseif not S.OptimizeCrystals then
restoreGlow()
end
end,
misc_optimize_crystals = function(v)

if v then
rescanWorldVisuals()
elseif not S.DisableCrystalsGlow then
restoreGlow()
end
end,
}
local KNOWN_OPTIONS = {
misc_transparent_textures_value = function(v)
v = tonumber(v)
if v then
S.TextureTransparency = math.clamp(v, 0, 1)
if S.TransparentTextures then rescanWorldVisuals() end
end
end,
}
local function setControlValue(id, value, silent)
UIState.values[id] = value
local fn = KNOWN_TOGGLES[id] or KNOWN_OPTIONS[id]
if fn then
pcall(fn, value)
elseif not silent then
notifyPending(id)
end
end

local function selectTab(id)
for k,p in pairs(UIState.pages) do p.Visible = (k==id) end
for k,b in pairs(UIState.tabButtons) do
b.BackgroundColor3 = (k==id) and COLORS.TileSelected or COLORS.Tile
local st=b:FindFirstChild("TabStroke")
if st then
st.Color=(k==id) and COLORS.Accent or Color3.new(1,1,1)
st.Transparency=(k==id) and 0.15 or 0.86
end
end
UIState.activeTab=id
end
