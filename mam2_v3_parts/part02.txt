local function crystalTitle(x)
local parent = x.Parent
local name = getAttr(x, "CrystalName")
or (parent and getAttr(parent, "CrystalName"))
or x.Name
local tier = getAttr(x, "TierName") or (parent and getAttr(parent, "TierName"))
local value = tonumber(getAttr(x, "Value") or (parent and getAttr(parent, "Value")))
local weight = tonumber(getAttr(x, "WeightKg") or (parent and getAttr(parent, "WeightKg")))
local luck = getAttr(x, "LuckBoost") or (parent and getAttr(parent, "LuckBoost"))
local meta = {}
if tier then meta[#meta+1] = tostring(tier) end
if value then meta[#meta+1] = "$" .. tostring(math.floor(value)) end
if weight then meta[#meta+1] = string.format("%.1fkg", weight) end
if luck then meta[#meta+1] = tostring(luck) end
return tostring(name), table.concat(meta, "  •  ")
end

local function httpGet(url)
local ok, body = pcall(function()
return game:HttpGet(url)
end)
if ok and type(body) == "string" then return body end
local req = request or http_request or (syn and syn.request)
if type(req) == "function" then
local ok2, res = pcall(req, {Url=url, Method="GET"})
if ok2 and type(res) == "table" then
return res.Body or res.body
end
end
end
local function serverList()
local url = string.format(
"https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100",
game.PlaceId
)
local body = httpGet(url)
if not body then return nil end
local ok, data = pcall(HttpService.JSONDecode, HttpService, body)
return ok and data or nil
end
local function hopServer(lowest)
task.spawn(function()
local data = serverList()
if not data or type(data.data) ~= "table" then
notify("Não foi possível consultar servidores.", 4)
return
end
local candidates = {}
for _, sv in ipairs(data.data) do
if sv.id ~= game.JobId
and tonumber(sv.playing)
and tonumber(sv.maxPlayers)
and sv.playing < sv.maxPlayers then
candidates[#candidates+1] = sv
end
end
if #candidates == 0 then
notify("Nenhum servidor alternativo encontrado.", 4)
return
end
if lowest then
table.sort(candidates, function(a,b)
return (a.playing or math.huge) < (b.playing or math.huge)
end)
end
local sv = candidates[1]
pcall(function()
TeleportService:TeleportToPlaceInstance(game.PlaceId, sv.id, LP)
end)
end)
end
